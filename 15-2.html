JavaScript 程序结构总览

JS 程序由三种基本结构组成：顺序结构（从上到下逐条执行）、分支结构（if / switch，根据条件选择执行路径）、循环结构（for / while / for-of 等，重复执行一段代码）。

// 程序结构示意
let zyt_score = 75;
// ① 顺序结构：逐条赋值、计算
let zyt_level = zyt_score + 10;
if (zyt_level >= 60) {
// ② 分支结构：条件满足才执行
console.log('及格');
} else {
console.log('不及格');
}
for (let zyt_i = 1; zyt_i <= 5; zyt_i++) {
// ③ 循环结构：重复执行 5 次
console.log('第' + zyt_i + '次');
}

for 循环（最常用）

语法：for (初始化; 条件; 步进) { 循环体 }。计算 1~10 的累加和。 

let zyt_sum = 0;
for (let zyt_i = 1; zyt_i <= 10; zyt_i++) {
  zyt_sum += zyt_i;  // 每次把 i 加到 sum 上
}
console.log('1 加到 10 =', zyt_sum);

while 循环（条件先判断）

语法：while (条件) { 循环体 }。输出 1~5 的平方。 

let zyt_j = 1;
while (zyt_j <= 5) {
  console.log(zyt_j + '² = ' + zyt_j * zyt_j);
  zyt_j++;  // 必须手动更新条件变量
}

do-while 循环（先执行后判断）

语法：do { 循环体 } while (条件);。无论条件是否成立，循环体至少执行一次。 

let zyt_k = 3;
do {
  console.log('倒计时: ' + zyt_k);
  zyt_k--;
} while (zyt_k > 0);
console.log('发射!');

for-of 循环（遍历数组）

语法：for (let 变量 of 数组) { }。ES6 新语法，直接拿到每个元素值，无需下标。 

const zyt_books = ['活着', '白夜行', '小王子', '围城'];
for (let zyt_name of zyt_books) {
  console.log('📖 ' + zyt_name);
}

for-in 循环（遍历对象键名）

语法：for (let key in 对象) { }。用于遍历对象的属性名（键）。 

const zyt_student = { name: '张语桐', age: 20, major: '高级网页设计' };
for (let zyt_key in zyt_student) {
  console.log(zyt_key + ' = ' + zyt_student[zyt_key]);
}

forEach 数组方法（函数式遍历）

语法：数组.forEach(function(item, index) { })。数组自带方法，回调函数拿到元素值和下标。 

const zyt_scores = [85, 92, 78, 95, 88];
zyt_scores.forEach(function (zyt_score, zyt_idx) {
  console.log('第' + (zyt_idx + 1) + '个分数: ' + zyt_score);
});
let zyt_avg = zyt_scores.reduce(function(a, b){return a + b;}, 0) / zyt_scores.length;
console.log('平均分: ' + zyt_avg.toFixed(1));
