  <script>
    const zytBookData = {
      hot: {
        feature: {

        },
        list: [
          { title: '活着', rating: 4.9, reviews: 4285, cover: 'img/22.jpg', current: 32, old: 49, badge: '满79减30', tag: '书友热评' },
          { title: '最好的我们', rating: 4.8, reviews: 1978, cover: 'img/23.jpg', current: 46, old: 71, badge: '满79减30' }
        ]
      }
    };

    const zytTabs = [
      { id: 'new', label: '新品特惠' },
      { id: 'week', label: '本周精选' },
      { id: 'hot', label: '读者热评' }
    ];

    function zytRenderStars(rating) {
      const full = Math.round(rating);
      let stars = '';
      for (let i = 1; i <= 5; i++) {
        stars += `<span class="${i <= full ? '' : 'off'}">★</span>`;
      }
      return stars;
    }

    function zytRenderHotTab() {
      const feature = zytBookData.hot.feature;
      const list = zytBookData.hot.list;
      const featureStars = zytRenderStars(feature.rating);

      document.getElementById('zyt_bookGrid').innerHTML = `
        <div class="zyt_hotLayout">
          <article class="zyt_featureBook">
            <div class="zyt_featureCover">
              <img src="${feature.cover}" alt="${feature.title}" />
              <span class="zyt_promoBadge">${feature.badge}</span>
            </div>
            <div class="zyt_featureBody">
              <div class="zyt_featureTitle">${feature.title}</div>
              <div class="zyt_featureDesc">${feature.subtitle}</div>
              <div class="zyt_featureMeta">
                <span class="zyt_stars">${featureStars}</span>
                <span>${feature.rating.toFixed(1)}</span>
                <span>(${feature.reviews})</span>
              </div>
              <div class="zyt_priceBox">
                <span class="zyt_priceLabel">预估到手价</span>
                <span class="zyt_currentPrice">¥${feature.current}</span>
                <span class="zyt_oldPrice">¥${feature.old}</span>
              </div>
              <div class="zyt_featureDesc" style="margin-top: 16px; margin-bottom: 0;">${feature.intro}</div>
            </div>
          </article>

          <div class="zyt_miniGrid">
            ${list.map(item => {
              const starHtml = zytRenderStars(item.rating);
              return `
                <article class="zyt_miniCard">
                  <div class="zyt_coverWrap">
                    <img src="${item.cover}" alt="${item.title}" />
                    <span class="zyt_promoBadge">${item.badge}</span>
                    ${item.tag ? `<span class="zyt_cardTag">${item.tag}</span>` : ''}
                  </div>
                  <div class="zyt_bookTitle">${item.title}</div>
                  <div class="zyt_ratingRow">
                    <span class="zyt_stars">${starHtml}</span>
                    <span>${item.rating.toFixed(1)}</span>
                    <span>(${item.reviews})</span>
                  </div>
                  <div class="zyt_priceBox">
                    <span class="zyt_priceLabel">预估到手价</span>
                    <span class="zyt_currentPrice">¥${item.current}</span>
                    <span class="zyt_oldPrice">¥${item.old}</span>
                  </div>
                </article>
              `;
            }).join('')}
          </div>
        </div>
      `;
    }

    zytTabs.forEach(tab => {
      const el = document.createElement('div');
