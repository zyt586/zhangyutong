// 用 setTimeout 把每一步包装成 Promise
function zyt_doStep(zyt_name, zyt_ms) {
  return new Promise(zyt_resolve => {
    setTimeout(() => zyt_resolve(zyt_name), zyt_ms);
  });
}

async function zyt_makeDumplings() {
  // ① 买原料：线上线下同时买，任一完成即可
  await Promise.race([
    zyt_doStep('线上买原料', 6000),
    zyt_doStep('线下买原料', 4000)
  ]);

  // ② 取原料
  await zyt_doStep('取原料', 2000);

  // ③ 和面、和馅同时进行，都完成才继续
  await Promise.all([
    zyt_doStep('和面', 5000),
    zyt_doStep('和馅', 3000)
  ]);

  // ④⑤⑥ 依次串行
  await zyt_doStep('包饺子', 4000);
  await zyt_doStep('煮饺子', 5000);
  await zyt_doStep('吃饺子', 1000);
}
